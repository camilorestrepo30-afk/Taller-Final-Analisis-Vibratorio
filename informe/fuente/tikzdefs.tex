\newcommand{\amort}[2]{%
  \draw[thick] (#1) -- ($(#1)!0.38!(#2)$);
  \draw[thick] ($(#1)!0.38!(#2)+(0,-0.16)$) -- ($(#1)!0.38!(#2)+(0,0.16)$);
  \draw[thick] ($(#1)!0.30!(#2)+(0,0.22)$) -- ($(#1)!0.62!(#2)+(0,0.22)$);
  \draw[thick] ($(#1)!0.30!(#2)+(0,-0.22)$) -- ($(#1)!0.62!(#2)+(0,-0.22)$);
  \draw[thick] ($(#1)!0.62!(#2)+(0,-0.22)$) -- ($(#1)!0.62!(#2)+(0,0.22)$);
  \draw[thick] ($(#1)!0.62!(#2)$) -- (#2);}
\newcommand{\pared}[3]{% x, ybottom, ytop
  \fill[pattern=north east lines] (#1-0.25,#2) rectangle (#1,#3);
  \draw[thick] (#1,#2) -- (#1,#3);}
\newcommand{\paredR}[3]{%
  \fill[pattern=north east lines] (#1,#2) rectangle (#1+0.25,#3);
  \draw[thick] (#1,#2) -- (#1,#3);}
